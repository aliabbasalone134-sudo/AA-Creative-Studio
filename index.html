/* ========================================================= LIVE NEWS — NETLIFY FUNCTION ========================================================= */ async function loadNews() { const newsList = document.getElementById("newsList"); if (!newsList) { return; } /* Loading message */ newsList.innerHTML = `
⏳
LIVE NEWS

Loading latest Pakistan news...

Please wait while the latest headlines are loaded.

LIVE
`; try { /* Connect to Netlify Function */ const response = await fetch("/.netlify/functions/news"); if (!response.ok) { throw new Error( "News server returned an error." ); } const data = await response.json(); /* Check API response */ if ( !data.success || !Array.isArray(data.news) || data.news.length === 0 ) { throw new Error( "No news available." ); } /* Remove loading message */ newsList.innerHTML = ""; /* Create news cards */ data.news.forEach(item => { const article = document.createElement("article"); article.className = "news-item"; /* News icon */ const image = document.createElement("div"); image.className = "news-image"; image.textContent = "📰"; /* Main content */ const content = document.createElement("div"); /* Source */ const category = document.createElement("div"); category.className = "news-category"; category.textContent = item.source || "Pakistan News"; /* Title */ const title = document.createElement("h4"); const link = document.createElement("a"); link.href = item.link || "#"; link.target = "_blank"; link.rel = "noopener noreferrer"; link.textContent = item.title || "Latest news"; title.appendChild(link); /* Description */ const description = document.createElement("p"); description.textContent = item.description || "Latest news from connected sources."; /* Add content */ content.appendChild(category); content.appendChild(title); content.appendChild(description); /* Date / time */ const time = document.createElement("div"); time.className = "news-time"; if (item.date) { const date = new Date(item.date); if (!isNaN(date.getTime())) { time.textContent = date.toLocaleDateString( "en-PK", { day:"2-digit", month:"short" } ); } else { time.textContent = "LIVE"; } } else { time.textContent = "LIVE"; } /* Build card */ article.appendChild(image); article.appendChild(content); article.appendChild(time); /* Add to news list */ newsList.appendChild(article); }); } catch (error) { console.error( "News loading error:", error ); /* Error message */ newsList.innerHTML = `
⚠️
NEWS

News could not be loaded.

Please try again later.

—
`; } } /* ========================================================= START LIVE NEWS ========================================================= */ loadNews(); /* ========================================================= AUTO REFRESH — EVERY 5 MINUTES ========================================================= */ setInterval( loadNews, 5 * 60 * 1000 );
